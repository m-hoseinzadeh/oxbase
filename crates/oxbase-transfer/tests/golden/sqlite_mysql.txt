# SQLite -> MySQL
boolean                  -> boolean                              exact
int8                     -> int64                                exact
int16                    -> int64                                exact
int32                    -> int64                                exact
int64                    -> int64                                exact
int32 unsigned           -> int64                                exact
int64 unsigned           -> text                                 exact
float32                  -> float64                              exact
float64                  -> float64                              exact
decimal(18,4)            -> decimal(18,4)                        exact
decimal(38,10)           -> decimal(38,10)                       exact
decimal                  -> decimal(65,30)                       narrowed    unconstrained decimals become decimal(65,30): values with more than 35 integer digits are refused, more than 30 decimal places are rounded
char(10)                 -> char(10)                             exact
varchar(255)             -> varchar(255)                         exact
varchar                  -> text                                 exact
text                     -> text                                 exact
binary(16)               -> varbinary                            exact
varbinary                -> varbinary                            exact
date                     -> date                                 exact
time(6)                  -> time(6)                              exact
timestamp(6)             -> timestamp(6)                         exact
timestamptz(6)           -> timestamp(6)                         narrowed    stored as datetime holding the UTC instant: the time zone offset is dropped
interval                 -> text                                 exact
uuid                     -> char(36)                             widened     MySQL has no uuid type: stored as its 36-character text
json                     -> json                                 exact
jsonb                    -> jsonb                                exact
int32[]                  -> text                                 exact
text[]                   -> text                                 exact
enum(small, medium, large) -> text                                 exact
geometry                 -> text                                 exact
