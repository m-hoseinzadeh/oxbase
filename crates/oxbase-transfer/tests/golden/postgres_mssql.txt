# PostgreSQL -> SQL Server
boolean                  -> boolean                              exact
int8                     -> int16                                exact
int16                    -> int16                                exact
int32                    -> int32                                exact
int64                    -> int64                                exact
int32 unsigned           -> int64                                exact
int64 unsigned           -> decimal(20,0)                        exact
float32                  -> float32                              exact
float64                  -> float64                              exact
decimal(18,4)            -> decimal(18,4)                        exact
decimal(38,10)           -> decimal(38,10)                       exact
decimal                  -> decimal(38,10)                       narrowed    unconstrained decimals become decimal(38,10): values with more than 28 integer digits are refused, more than 10 decimal places are refused
char(10)                 -> char(10)                             exact
varchar(255)             -> varchar(255)                         exact
varchar                  -> text                                 exact
text                     -> text                                 exact
binary(16)               -> varbinary                            exact
varbinary                -> varbinary                            exact
date                     -> date                                 exact
time(6)                  -> time(6)                              exact
timestamp(6)             -> timestamp(6)                         exact
timestamptz(6)           -> timestamptz(6)                       exact
interval                 -> varchar(64)                          widened     SQL Server has no interval type: stored as the source's text form
uuid                     -> uuid                                 exact
json                     -> json                                 exact
jsonb                    -> jsonb                                exact
int32[]                  -> nvarchar(max) [json]                 widened     SQL Server has no array types: stored as JSON arrays in nvarchar(max)
text[]                   -> nvarchar(max) [json]                 widened     SQL Server has no array types: stored as JSON arrays in nvarchar(max)
enum(small, medium, large) -> text                                 exact
geometry                 -> geometry                             unsupported PostgreSQL has no column type for geometry
